1. Plain HTML

<link rel="stylesheet" href="digestive-system.css">
<script type="module" src="digestive-system.js"></script>

<digestive-system interactive labels caption lang="ar"></digestive-system>

<script type="module">
  document.querySelector('digestive-system')
    .addEventListener('organ-select', e => console.log(e.detail.id));
</script>

2. React / Next.js

import { useEffect, useRef } from 'react';
import 'digestive-system/digestive-system.css';

export function DigestiveSystem({ onSelect, lang = 'en' }) {
  const ref = useRef(null);
  useEffect(() => { import('digestive-system/digestive-system.js'); }, []);
  useEffect(() => {
    const el = ref.current;
    const h = e => onSelect?.(e.detail);
    el.addEventListener('organ-select', h);
    return () => el.removeEventListener('organ-select', h);
  }, [onSelect]);
  return <digestive-system ref={ref} interactive caption lang={lang} />;
}

In Next.js load it from a client component ('use client'); custom elements need the DOM.

5. Events & API

el.select('large-intestine');   // programmatic selection
el.clear();                     // drop hover + selection
el.selected;                    // 'large-intestine' | null
el.organs;                      // [{ id, en, ar, x, y }, ...]

el.addEventListener('organ-hover',  e => e.detail.id);
el.addEventListener('organ-select', e => e.detail.name);

Organ ids: oesophagus, stomach, duodenum, small-intestine, large-intestine, appendix, rectum.
